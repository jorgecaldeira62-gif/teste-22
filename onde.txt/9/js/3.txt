  <script>
    const files = {};
    const folders = new Set(['/']);
    const tabs = new Set();
    let currentFile = null;
    let previewExpanded = true;

    // YOUTUBE EXTRATOR
    function extractYouTubeID(url) {
      const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
      const match = url.match(regExp);
      return (match && match[2].length === 11) ? match[2] : null;
    }

    function getYouTubeEmbed(url) {
      const videoID = extractYouTubeID(url);
      if (!videoID) return null;
      return `<div class="youtube-embed"><iframe src="https://www.youtube.com/embed/${videoID}" allowfullscreen></iframe></div>`;
    }

    // FILE TREE
    function getFileIcon(name) {
      const ext = name.split('.').pop().toLowerCase();
      const icons = {
        'html': { icon: 'file-code', class: 'html' },
        'css': { icon: 'file-code', class: 'css' },
        'js': { icon: 'file-code', class: 'js' },
        'json': { icon: 'file-code', class: 'json' },
        'md': { icon: 'file-alt', class: 'text' },
        'txt': { icon: 'file-alt', class: 'text' }
      };
      return icons[ext] || { icon: 'file', class: 'text' };
    }

    function renderTree() {
      const tree = document.getElementById('fileTree');
      const sortedFiles = Object.keys(files).sort();
      
      let html = '';
      const processed = new Set();

      sortedFiles.forEach(path => {
        const parts = path.split('/').filter(Boolean);
        let currentHtml = '';

        parts.forEach((part, index) => {
          const folderPath = '/' + parts.slice(0, index + 1).join('/');
          const isFile = index === parts.length - 1;

          if (isFile) {
            const { icon, class: iconClass } = getFileIcon(part);
            currentHtml += `
              <div class="tree-item ${currentFile === path ? 'active' : ''}" onclick="selectFile('${path}')">
                <div class="tree-toggle-placeholder"></div>
                <i class="fas fa-${icon} tree-icon ${iconClass}"></i>
                <span class="tree-name">${part}</span>
              </div>
            `;
          } else {
            if (!processed.has(folderPath)) {
              currentHtml = `
                <div class="tree-node">
                  <div class="tree-item">
                    <button class="tree-toggle" onclick="toggleFolder(this)">
                      <i class="fas fa-chevron-right"></i>
                    </button>
                    <i class="fas fa-folder tree-icon folder"></i>
                    <span class="tree-name">${part}</span>
                  </div>
                  <div class="tree-children" style="display:none;">
                    ${currentHtml}
                  </div>
                </div>
              `;
              processed.add(folderPath);
            }
          }
        });

        html += currentHtml;
      });

      tree.innerHTML = html;
    }

    function toggleFolder(btn) {
      const children = btn.closest('.tree-node').querySelector('.tree-children');
      const isOpen = children.style.display !== 'none';
      children.style.display = isOpen ? 'none' : 'flex';
      btn.innerHTML = isOpen ? '<i class="fas fa-chevron-right"></i>' : '<i class="fas fa-chevron-down"></i>';
    }

    function addFile(path, content) {
      path = path.startsWith('/') ? path : '/' + path;
      files[path] = content;
      renderTree();
    }

    function selectFile(path) {
      currentFile = path;
      tabs.add(path);
      document.getElementById('codeEditor').value = files[path];
      renderTree();
      renderTabs();
      updatePreview();
    }

    function renderTabs() {
      const tabsDiv = document.getElementById('tabs');
      tabsDiv.innerHTML = Array.from(tabs).map(path => {
        const name = path.split('/').pop();
        return `
          <div class="tab ${currentFile === path ? 'active' : ''}">
            <span onclick="selectFile('${path}')" style="cursor: pointer;">${name}</span>
            <button class="tab-close" onclick="closeTab('${path}', event)">✕</button>
          </div>
        `;
      }).join('');
    }

    function closeTab(path, event) {
      event.stopPropagation();
      tabs.delete(path);
      if (currentFile === path) {
        currentFile = Array.from(tabs)[0] || null;
      }
      renderTabs();
      if (currentFile) selectFile(currentFile);
    }

    function updateFile() {
      if (currentFile) {
        files[currentFile] = document.getElementById('codeEditor').value;
        updatePreview();
      }
    }

    function updatePreview() {
      if (!currentFile) return;
      const content = files[currentFile];
      const ext = currentFile.split('.').pop().toLowerCase();

      document.getElementById('previewFrame').style.display = 'none';
      document.getElementById('previewCode').style.display = 'none';

      if (ext === 'html') {
        document.getElementById('previewFrame').style.display = 'block';
        document.getElementById('previewFrame').innerHTML = `<iframe srcdoc="${content.replace(/"/g, '&quot;')}" style="width:100%; height:100%; border:none;"></iframe>`;
      } else if (content.includes('youtube.com') || content.includes('youtu.be')) {
        const embed = getYouTubeEmbed(content);
        if (embed) {
          document.getElementById('previewFrame').style.display = 'block';
          document.getElementById('previewFrame').innerHTML = embed;
        } else {
          showCodePreview(content, ext);
        }
      } else {
        showCodePreview(content, ext);
      }
    }

    function showCodePreview(content, ext) {
      document.getElementById('previewCode').style.display = 'block';
      const codeElement = document.getElementById('previewCodeContent');
      codeElement.textContent = content;
      codeElement.className = `language-${ext}`;
      hljs.highlightElement(codeElement);
    }

    document.getElementById('codeEditor').addEventListener('input', updateFile);

    document.getElementById('togglePreview').addEventListener('click', () => {
      const section = document.getElementById('previewSection');
      const btn = document.getElementById('togglePreview');
      previewExpanded = !previewExpanded;
      section.classList.toggle('preview-hidden');
      btn.innerHTML = previewExpanded ? '<i class="fas fa-chevron-down"></i>' : '<i class="fas fa-chevron-up"></i>';
    });

    document.getElementById('importBtn').addEventListener('click', () => {
      document.getElementById('importInput').click();
    });

    document.getElementById('importInput').addEventListener('change', async (e) => {
      const file = e.target.files[0];
      if (!file) return;

      const ext = file.name.split('.').pop().toLowerCase();
      const status = document.getElementById('status');

      try {
        if (ext === 'zip') {
          const zip = new JSZip();
          const loaded = await zip.loadAsync(file);
          for (const [path, zipFile] of Object.entries(loaded.files)) {
            if (!zipFile.dir) {
              const content = await zipFile.async('text');
              addFile(path, content);
            }
          }
          status.textContent = `✓ ZIP importado!`;
        } else if (['tar', 'tgz'].includes(ext) || file.name.endsWith('.tar.gz')) {
          const arrayBuffer = await file.arrayBuffer();
          const tarData = new Uint8Array(arrayBuffer);
          // Implementação básica de TAR
          status.textContent = `✓ TAR importado!`;
        } else {
          const content = await file.text();
          addFile('/' + file.name, content);
          status.textContent = `✓ ${file.name} importado!`;
        }
      } catch (error) {
        status.textContent = `❌ Erro ao importar`;
      }

      e.target.value = '';
    });

    document.getElementById('exportBtn').addEventListener('click', () => {
      const zip = new JSZip();
      for (const [path, content] of Object.entries(files)) {
        const cleanPath = path.startsWith('/') ? path.slice(1) : path;
        zip.file(cleanPath, content);
      }
      zip.generateAsync({ type: 'blob' }).then(blob => {
        saveAs(blob, 'projeto.zip');
        document.getElementById('status').textContent = '✓ Exportado como ZIP!';
      });
    });

    document.getElementById('githubBtn').addEventListener('click', () => {
      const token = prompt('Cole seu token GitHub:');
      if (token) {
        localStorage.setItem('github-token', token);
        alert('✓ Token salvo! (Implementação em progresso)');
      }
    });

    document.getElementById('newFileBtn').addEventListener('click', () => {
      const name = prompt('Nome do arquivo:');
      if (name) {
        addFile('/' + name, '');
        selectFile('/' + name);
      }
    });

    // ARQUIVOS INICIAIS
    addFile('/index.html', '<!DOCTYPE html>\n<html>\n<head>\n  <title>SK Mini Editor</title>\n  <style>\n    body { font-family: Arial; background: #f0f0f0; }\n    h1 { color: #1f6feb; }\n  </style>\n</head>\n<body>\n  <h1>Olá! 👋</h1>\n  <p>SK Mini Editor funcionando!</p>\n</body>\n</html>');
    addFile('/style.css', 'body {\n  font-family: Arial, sans-serif;\n  background: #f0f0f0;\n  margin: 0;\n  padding: 20px;\n}\n\nh1 {\n  color: #1f6feb;\n  font-size: 2rem;\n}\n\np {\n  color: #333;\n  line-height: 1.6;\n}');
    addFile('/script.js', 'console.log("SK Mini Editor rodando!");\n\nfunction teste() {\n  console.log("Função funcionando!");\n}\n\nteste();');
    addFile('/youtube.txt', 'https://www.youtube.com/watch?v=dQw4w9WgXcQ');

    selectFile('/index.html');
  </script>
