<html>
<head>
  <title>SK Code Editor</title>
  <script src="https://cdn.tailwindcss.com"></script>
</head>
<body>
  <div id="root"></div>

  <script type="module">
    import React from 'react';
    import ReactDOM from 'react-dom/client';
    import GitHubPanel from './GitHubPanel';

    // Seus arquivos do projeto
    const files = {
      'index.html': '<h1>Hello</h1>',
      'style.css': 'body { color: blue; }',
      'script.js': 'console.log("test");'
    };

    function App() {
      const [projectFiles, setProjectFiles] = React.useState(files);

      return (
        <div style={{ display: 'flex', height: '100vh' }}>
          <div style={{ flex: 1 }}>
            {/* Seu editor aqui */}
          </div>
          <div style={{ width: '350px', borderLeft: '1px solid #ccc' }}>
            <GitHubPanel
              files={projectFiles}
              onImport={(importedFiles) => setProjectFiles(importedFiles)}
              projectName="Meu Projeto"
            />
          </div>
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById('root')).render(<App />);
  </script>
</body>
