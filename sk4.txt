  <script>
    const files = {};
    const tabs = new Set();
    let currentFile = null;

    function addFile(name, content) {
      files[name] = content;
      renderFiles();
    }

    function renderFiles() {
      const list = document.getElementById('fileList');
      list.innerHTML = Object.keys(files).map(name => {
        const ext = name.split('.').pop();
        let icon = 'file';
        let iconClass = '';
        if (ext === 'html') { icon = 'file-code'; iconClass = 'html'; }
        else if (ext === 'css') { icon = 'file-code'; iconClass = 'css'; }
        else if (ext === 'js') { icon = 'file-code'; iconClass = 'js'; }
        
        return `
          <div class="file-item ${currentFile === name ? 'active' : ''}" onclick="selectFile('${name}')">
            <i class="fas fa-${icon} file-icon ${iconClass}"></i>
            <span>${name}</span>
          </div>
        `;
      }).join('');
    }

    function selectFile(name) {
      currentFile = name;
      tabs.add(name);
      document.getElementById('editor').value = files[name];
      renderFiles();
      renderTabs();
      updatePreview();
    }

    function renderTabs() {
      const tabsDiv = document.getElementById('tabs');
      tabsDiv.innerHTML = Array.from(tabs).map(name => `
        <div class="tab ${currentFile === name ? 'active' : ''}">
          <span onclick="selectFile('${name}')">${name}</span>
          <button class="tab-close" onclick="closeTab('${name}')">✕</button>
        </div>
      `).join('');
    }

    function closeTab(name) {
      tabs.delete(name);
      if (currentFile === name) {
        currentFile = Array.from(tabs)[0] || null;
      }
      renderTabs();
      if (currentFile) selectFile(currentFile);
    }

    function updateFile() {
      if (currentFile) {
        files[currentFile] = document.getElementById('editor').value;
        updatePreview();
      }
    }

    function updatePreview() {
      if (!currentFile) return;
      const content = files[currentFile];
      const ext = currentFile.split('.').pop();
      
      document.getElementById('previewFrame').style.display = 'none';
      document.getElementById('previewImage').style.display = 'none';
      document.getElementById('previewText').style.display = 'none';

      if (ext === 'html') {
        document.getElementById('previewFrame').style.display = 'block';
        document.getElementById('previewFrame').srcDoc = content;
      } else {
        document.getElementById('previewText').style.display = 'block';
        document.getElementById('previewText').innerHTML = `<pre>${content.replace(/</g, '&lt;').replace(/>/g, '&gt;')}</pre>`;
      }
    }

    document.getElementById('editor').addEventListener('input', updateFile);

    document.getElementById('importBtn').addEventListener('click', () => {
      document.getElementById('importInput').click();
    });

    document.getElementById('importInput').addEventListener('change', async (e) => {
      const file = e.target.files[0];
      if (!file) return;

      const ext = file.name.split('.').pop().toLowerCase();
      
      if (ext === 'zip') {
        const zip = new JSZip();
        const loaded = await zip.loadAsync(file);
        for (const [path, zipFile] of Object.entries(loaded.files)) {
          if (!zipFile.dir) {
            const content = await zipFile.async('text');
            addFile(path.split('/').pop(), content);
          }
        }
      } else {
        const content = await file.text();
        addFile(file.name, content);
      }

      document.getElementById('status').textContent = `✓ ${file.name} importado!`;
      e.target.value = '';
    });

    document.getElementById('exportBtn').addEventListener('click', () => {
      const zip = new JSZip();
      for (const [name, content] of Object.entries(files)) {
        zip.file(name, content);
      }
      zip.generateAsync({ type: 'blob' }).then(blob => {
        saveAs(blob, 'projeto.zip');
        document.getElementById('status').textContent = '✓ Exportado como ZIP!';
      });
    });

    addFile('index.html', '<!DOCTYPE html>\n<html>\n<head>\n  <title>Teste</title>\n</head>\n<body>\n  <h1>Olá!</h1>\n  <p>SK Mini Editor funcionando!</p>\n</body>\n</html>');
    addFile('style.css', 'body {\n  font-family: Arial;\n  background: #f0f0f0;\n}\n\nh1 {\n  color: blue;\n}');
    addFile('script.js', 'console.log("Funcionando!");\nalert("SK Mini Editor está rodando!");');

    selectFile('index.html');
  </script>
